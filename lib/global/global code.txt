import 'dart:math';

double simpleFare(LatLng pickup, LatLng dropoff) {
  const double baseFare = 5.0; // fixed starting fare
  const double perKm = 2.0; // rate per km

  double toRadians(double degree) => degree * pi / 180;

  double lat1 = pickup.latitude;
  double lon1 = pickup.longitude;
  double lat2 = dropoff.latitude;
  double lon2 = dropoff.longitude;

  double dLat = toRadians(lat2 - lat1);
  double dLon = toRadians(lon2 - lon1);

  double a = sin(dLat / 2) * sin(dLat / 2) +
      cos(toRadians(lat1)) *
          cos(toRadians(lat2)) *
          sin(dLon / 2) *
          sin(dLon / 2);
  double c = 2 * atan2(sqrt(a), sqrt(1 - a));
  const R = 6371; // Earth radius in km
  double distanceKm = R * c;

  double fare = baseFare + (distanceKm * perKm);
  return double.parse(fare.toStringAsFixed(2)); // rounded to 2 decimals
}

class _DriverHomeScreenState extends State<DriverHomeScreen> {
  final FirebaseService _fs = FirebaseService();
  final Location _location = Location();
  GoogleMapController? _mapController;
  LatLng? _currentLocation;

  @override
  void initState() {
    super.initState();
    _initLocation();
  }

  Future<void> _initLocation() async {
    final loc = await _location.getLocation();
    setState(() {
      _currentLocation = LatLng(loc.latitude!, loc.longitude!);
    });

    _location.onLocationChanged.listen((locData) {
      if (_mapController != null) {
        final newLoc = LatLng(locData.latitude!, locData.longitude!);
        _mapController!.animateCamera(CameraUpdate.newLatLng(newLoc));
        _fs.updateDriverLocation(
          driverId: 'driver_test_1',
          lat: locData.latitude!,
          lng: locData.longitude!,
        );
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: _currentLocation == null
          ? Center(child: CircularProgressIndicator())
          : GoogleMap(
              initialCameraPosition:
                  CameraPosition(target: _currentLocation!, zoom: 14),
              myLocationEnabled: true,
              myLocationButtonEnabled: true,
            ),
    );
  }
}

class _DriverRequestListScreenState extends State<DriverRequestListScreen> {
  final FirebaseService _fs = FirebaseService();
  Ride? _acceptedRide;
  LocationData? _driverLocation;
  late final Location _location;

  @override
  void initState() {
    super.initState();
    _location = Location();
    _initLocation();
  }

  Future<void> _initLocation() async {
    final hasPermission = await _location.requestPermission();
    if (hasPermission == PermissionStatus.granted) {
      _location.onLocationChanged.listen((loc) {
        setState(() => _driverLocation = loc);
        if (_acceptedRide != null) {
          _fs.updateDriverLocation(
            driverId: _acceptedRide!.id,
            lat: loc.latitude!,
            lng: loc.longitude!,
          );
        }
      });
    }
  }

  Future<String> _formatAddress(Map<String, dynamic> coords) async {
    try {
      final placemarks = await geocoding.placemarkFromCoordinates(
        coords['lat'],
        coords['lng'],
      );
      if (placemarks.isNotEmpty) {
        final p = placemarks.first;
        return "${p.street}, ${p.locality}";
      }
    } catch (_) {}
    return "${coords['lat']}, ${coords['lng']}";
  }

  Future<void> _updateRideStatus(String status) async {
    if (_acceptedRide != null) {
      await _fs.updateRide(_acceptedRide!.id, {
        'status': status,
      });
      setState(() {
        _acceptedRide = Ride(
          riderId: 'riderID: hardcoded',
          id: _acceptedRide!.id,
          pickup: _acceptedRide!.pickup,
          dropoff: _acceptedRide!.dropoff,
          status: status,
          driverId: _acceptedRide!.driverId,
          fare: 5,
          timestamp: DateTime.now(),
          paymentStatus: 'unpaid',
        );
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_acceptedRide == null) {
      // Show available ride requests
      return Scaffold(
        appBar: AppBar(title: const Text('Available Requests')),
        body: StreamBuilder<List<Ride>>(
          stream: _fs.streamAvailableRequests(),
          builder: (context, snapshot) {
            if (!snapshot.hasData) {
              return const Center(child: CircularProgressIndicator());
            }
            final rides = snapshot.data!;
            if (rides.isEmpty) {
              return const Center(child: Text('No ride requests'));
            }
            return ListView.builder(
              itemCount: rides.length,
              itemBuilder: (context, index) {
                final ride = rides[index];
                return FutureBuilder<String>(
                  future: _formatAddress(ride.pickup),
                  builder: (context, pickupSnap) {
                    return FutureBuilder<String>(
                      future: _formatAddress(ride.dropoff),
                      builder: (context, dropoffSnap) {
                        return ListTile(
                          title: Text(
                            'Pickup: ${pickupSnap.data ?? '...'}',
                          ),
                          subtitle: Text(
                            'Dropoff: ${dropoffSnap.data ?? '...'}',
                          ),
                          trailing: ElevatedButton(
                            onPressed: () async {
                              await _fs.updateRide(ride.id, {
                                'status': 'accepted',
                                'driverId':
                                    'driver_test_1', // TODO: replace with FirebaseAuth uid
                              });
                              setState(() {
                                _acceptedRide = Ride(
                                    riderId: 'rider_test_1',
                                    id: ride.id,
                                    pickup: ride.pickup,
                                    dropoff: ride.dropoff,
                                    status: 'accepted',
                                    driverId: 'driver_test_1',
                                    fare: 5,
                                    timestamp: DateTime.now(),
                                    paymentStatus: 'unpaid');
                              });
                            },
                            child: const Text('Accept'),
                          ),
                        );
                      },
                    );
                  },
                );
              },
            );
          },
        ),
      );
    } else {
      // Show active ride map + workflow buttons
      return Scaffold(
        appBar: AppBar(title: const Text('Active Ride')),
        body: Column(
          children: [
            Expanded(
              child: GoogleMap(
                initialCameraPosition: CameraPosition(
                  target: LatLng(
                    _driverLocation?.latitude ?? _acceptedRide!.pickup['lat'],
                    _driverLocation?.longitude ?? _acceptedRide!.pickup['lng'],
                  ),
                  zoom: 14,
                ),
                markers: {
                  Marker(
                    markerId: const MarkerId('pickup'),
                    position: LatLng(
                      _acceptedRide!.pickup['lat'],
                      _acceptedRide!.pickup['lng'],
                    ),
                    infoWindow: const InfoWindow(title: 'Pickup'),
                  ),
                  Marker(
                    markerId: const MarkerId('dropoff'),
                    position: LatLng(
                      _acceptedRide!.dropoff['lat'],
                      _acceptedRide!.dropoff['lng'],
                    ),
                    infoWindow: const InfoWindow(title: 'Dropoff'),
                  ),
                  if (_driverLocation != null)
                    Marker(
                      markerId: const MarkerId('driver'),
                      position: LatLng(
                        _driverLocation!.latitude!,
                        _driverLocation!.longitude!,
                      ),
                      infoWindow: const InfoWindow(title: 'You'),
                      icon: BitmapDescriptor.defaultMarkerWithHue(
                        BitmapDescriptor.hueAzure,
                      ),
                    ),
                },
              ),
            ),
            Padding(
              padding: const EdgeInsets.all(8.0),
              child: Column(
                children: [
                  Text('Ride Status: ${_acceptedRide!.status}'),
                  const SizedBox(height: 10),
                  if (_acceptedRide!.status == 'accepted')
                    ElevatedButton(
                      onPressed: () => _updateRideStatus('arrived'),
                      child: const Text('Mark Arrived'),
                    ),
                  if (_acceptedRide!.status == 'arrived')
                    ElevatedButton(
                      onPressed: () => _updateRideStatus('in_progress'),
                      child: const Text('Start Trip'),
                    ),
                  if (_acceptedRide!.status == 'in_progress')
                    ElevatedButton(
                      onPressed: () => _updateRideStatus('completed'),
                      child: const Text('Complete Trip'),
                    ),
                  if (_acceptedRide!.status == 'completed')
                    ElevatedButton(
                      onPressed: () {
                        setState(() => _acceptedRide = null);
                      },
                      child: const Text('Finish & Go Back'),
                    ),
                ],
              ),
            ),
          ],
        ),
      );
    }
  }

import 'package:cloud_firestore/cloud_firestore.dart';

class Ride {
  String id;
  String riderId;
  String? driverId;
  Map<String, dynamic> pickup;
  Map<String, dynamic> dropoff;
  String status;
  double fare;
  DateTime timestamp;
  String paymentStatus;

  Ride({
    required this.id,
    required this.riderId,
    this.driverId,
    required this.pickup,
    required this.dropoff,
    required this.status,
    required this.fare,
    required this.timestamp,
    required this.paymentStatus,
  });

  Map<String, dynamic> toMap() => {
        'id': id,
        'riderId': riderId,
        'driverId': driverId,
        'pickup': pickup,
        'dropoff': dropoff,
        'status': status,
        'fare': fare,
        'timestamp': timestamp,
        'paymentStatus': paymentStatus,
      };

  static Ride fromMap(Map<String, dynamic> m) => Ride(
        id: m['id'],
        riderId: m['riderId'],
        driverId: m['driverId'],
        pickup: Map<String, dynamic>.from(m['pickup']),
        dropoff: Map<String, dynamic>.from(m['dropoff']),
        status: (m['status'] ?? 'requested').toString().toLowerCase(),
        fare: (m['fare'] ?? 0).toDouble(),
        timestamp: (m['timestamp'] is Timestamp)
            ? (m['timestamp'] as Timestamp).toDate()
            : DateTime.tryParse(m['timestamp'].toString()) ?? DateTime.now(),
        paymentStatus: m['paymentStatus'] ?? 'unpaid',
      );
}

class _MoMoPaymentWidgetState extends State<MoMoPaymentWidget> {
  final TextEditingController phoneController = TextEditingController();
  bool isLoading = false;
  String? statusMessage;

  Future<void> _payWithMoMo() async {
    final phone = phoneController.text.trim();

    if (phone.isEmpty) {
      setState(() => statusMessage = "Please enter your phone number");
      return;
    }

    if (!phone.startsWith('7') || phone.length != 8) {
      setState(
          () => statusMessage = "Please enter a valid MTN number (7xxxxxxx)");
      return;
    }

    setState(() {
      isLoading = true;
      statusMessage = null;
    });

    try {
      final HttpsCallable callable =
          FirebaseFunctions.instance.httpsCallable('initiateMoMoPayment');
      final result = await callable.call({
        'rideId': widget.rideId,
        'phoneNumber': phone,
        'amount': widget.amount,
      });

      setState(() {
        statusMessage = "Payment request sent! Approve on your phone.";
      });

      // Call the success callback
      widget.onPaymentSuccess();
    } catch (e) {
      setState(() {
        statusMessage = "Payment failed: $e";
      });
    } finally {
      setState(() => isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.all(16),
      decoration: const BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          const SizedBox(height: 8),
          Container(
            width: 40,
            height: 4,
            decoration: BoxDecoration(
              color: Colors.grey[300],
              borderRadius: BorderRadius.circular(2),
            ),
          ),
          const SizedBox(height: 16),
          const Text(
            "Pay with MTN MoMo",
            style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
          ),
          const SizedBox(height: 8),
          Text(
            "Amount: E ${widget.amount.toStringAsFixed(2)}",
            style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold),
          ),
          const SizedBox(height: 16),
          TextField(
            controller: phoneController,
            keyboardType: TextInputType.phone,
            decoration: const InputDecoration(
              labelText: "Enter your phone (7xxxxxxx)",
              border: OutlineInputBorder(),
            ),
            inputFormatters: [
              FilteringTextInputFormatter.digitsOnly,
              LengthLimitingTextInputFormatter(8),
            ],
          ),
          const SizedBox(height: 16),
          ElevatedButton(
            onPressed: isLoading ? null : _payWithMoMo,
            style: ElevatedButton.styleFrom(

            ),
            child: isLoading
                ? const CircularProgressIndicator(color: Colors.white)
                : const Text("Pay Now"),
          ),

                  color: statusMessage!.startsWith("Payment failed")

            ),
          ],
        ],
      ),
    );
  }

Future<String?> requestMomoPayment({
  required String phone,
  required double amount,
  required String rideId,
}) async {
  try {
    final callable = FirebaseFunctions.instanceFor(region: 'us-central1')
        .httpsCallable('requestMomoPayment');

    final result = await callable.call(<String, dynamic>{
      'phone': phone,
      'amount': amount,
      'rideId': rideId,
    });

    return result.data['referenceId'] as String?;
  } catch (e) {
    print('Error requesting MoMo payment: $e');
    return null;
  }
}

void listenForPayment(String rideId) {
  FirebaseFirestore.instance
      .collection('rides')
      .doc(rideId)
      .snapshots()
      .listen((snapshot) {
    if (!snapshot.exists) return;

    final data = snapshot.data();
    final paid = data?['paid'] as bool? ?? false;
    final status = data?['momoStatus'] as String? ?? 'PENDING';

    if (paid) {
      print('Payment successful for ride $rideId!');
    } else {
      print('Payment status: $status');
    }
  });
}

class _HistoryScreenState extends State<HistoryScreen> {
  final User? user = FirebaseAuth.instance.currentUser;
  late Stream<QuerySnapshot> _ridesStream;

  @override
  void initState() {
    super.initState();
    _ridesStream = FirebaseFirestore.instance
        .collection('rides_history')
        .where('riderId', isEqualTo: user!.uid)
        .orderBy('timestamp', descending: true)
        .snapshots();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      extendBodyBehindAppBar: true,
      backgroundColor: Colors.transparent,
      appBar: AppBar(
        title: const Text(
          'Ride History',
          style: TextStyle(
            fontWeight: FontWeight.bold,
            fontSize: 22,
          ),
        ),
        backgroundColor: Colors.transparent,
        elevation: 0,
        foregroundColor: Colors.deepPurple,
      ),
      body: Container(
        decoration: BoxDecoration(
          gradient: LinearGradient(
            colors: [Colors.deepPurple.shade600, Colors.purple.shade50],
            begin: Alignment.topCenter,
            end: Alignment.bottomCenter,
          ),
        ),
        child: StreamBuilder<QuerySnapshot>(
          stream: _ridesStream,
          builder: (context, snapshot) {
            if (snapshot.hasError) {
              return Center(
                child: Text('Error: ${snapshot.error}'),
              );
            }
            if (snapshot.connectionState == ConnectionState.waiting) {
              return const Center(
                child: CircularProgressIndicator(),
              );
            }
            final docs = snapshot.data!.docs;
            if (docs.isEmpty) {
              return Center(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Icon(Icons.history, size: 80, color: Colors.grey.shade300),
                    const SizedBox(height: 16),
                    Text(
                      'No ride history yet',
                      style: TextStyle(
                        fontSize: 18,
                        color: Colors.black,
                      ),
                    ),
                  ],
                ),
              );
            }

            return ListView.separated(
              padding: const EdgeInsets.fromLTRB(16, 120, 16, 24),
              physics: const BouncingScrollPhysics(),
              itemCount: docs.length,
              separatorBuilder: (_, __) => const SizedBox(height: 16),
              itemBuilder: (context, index) {
                final ride = docs[index].data()! as Map<String, dynamic>;
                return _buildRideCard(ride);
              },
            );
          },
        ),
      ),
    );
  }

  Widget _buildRideCard(Map<String, dynamic> ride) {
    final timestamp = (ride['timestamp'] as Timestamp).toDate();
    final dateLabel = DateFormat.yMMMMd().format(timestamp);

    super.initState();

  Future<void> _fetchUserData() async {
    if (user != null) {
      final doc = await FirebaseFirestore.instance
          .collection('users')
          .doc(user!.uid)
          .get();
      if (doc.exists) {
        setState(() {
          userData = doc.data()!;
        });
      }
    }
  }

  Future<void> _pickAndUploadImage() async {
    try {
      final XFile? pickedFile = await _imagePicker.pickImage(
        source: ImageSource.gallery,
        imageQuality: 85,
        maxWidth: 512,
        maxHeight: 512,
      );

      if (pickedFile != null) {
        setState(() {
          _isUploading = true;
        });

        // Upload to Firebase Storage
        final File imageFile = File(pickedFile.path);
        final String fileName =
            'profile_${user!.uid}_${DateTime.now().millisecondsSinceEpoch}.jpg';

        // Upload to Firebase Storage
        final Reference storageRef = FirebaseStorage.instance
            .ref()
            .child('profile_pictures')
            .child(fileName);

        final UploadTask uploadTask = storageRef.putFile(imageFile);
        final TaskSnapshot snapshot = await uploadTask;

        // Get download URL
        final String downloadUrl = await snapshot.ref.getDownloadURL();

        // Update Firestore with the new profile picture URL
        await FirebaseFirestore.instance
            .collection('users')
            .doc(user!.uid)
            .update({
          'profilePictureUrl': downloadUrl,
          'updatedAt': FieldValue.serverTimestamp(),
        });

        // Refresh user data
        await _fetchUserData();

        CommonMethods.displaySnackBar(
            'Profile picture updated successfully!', context);
      }
    } catch (e) {
      CommonMethods.displaySnackBar('Error uploading image: $e', context);
    } finally {
      setState(() {
        _isUploading = false;
      });
    }
  }

  Future<void> _removeProfilePicture() async {
    try {
      setState(() {
        _isUploading = true;
      });

      // Remove from Firestore
      await FirebaseFirestore.instance
          .collection('users')
          .doc(user!.uid)
          .update({
        'profilePictureUrl': FieldValue.delete(),
        'updatedAt': FieldValue.serverTimestamp(),
      });

      // Refresh user data
      await _fetchUserData();

      CommonMethods.displaySnackBar('Profile picture removed', context);
    } catch (e) {
      CommonMethods.displaySnackBar(
          'Error removing profile picture: $e', context);
    } finally {
      setState(() {
        _isUploading = false;
      });
    }
  }

  }

  @override

    if (state == AppLifecycleState.paused) {
      _location.changeSettings(interval: 0);
      _locationSubscription?.pause();
    } else if (state == AppLifecycleState.resumed) {
      _location.changeSettings(interval: 1000);
      _locationSubscription?.resume();
    }
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    _mapController?.dispose();
    pickupController.dispose();
    dropoffController.dispose();
    _locationSubscription?.cancel();
    super.dispose();
  }

  Future<void> _initLocation() async {
    try {
      bool serviceEnabled = await _location.serviceEnabled();
      if (!serviceEnabled) {
        serviceEnabled = await _location.requestService();
        if (!serviceEnabled) {
          CommonMethods.displaySnackBar(
              'Location service is required', context);
          return;
        }
      }

      PermissionStatus permissionGranted = await _location.hasPermission();
      if (permissionGranted == PermissionStatus.denied) {
        permissionGranted = await _location.requestPermission();
        if (permissionGranted != PermissionStatus.granted) {
          CommonMethods.displaySnackBar('Location permission denied', context);
          return;
        }
      }

      final loc = await _location.getLocation();
      if (loc.latitude == null || loc.longitude == null) {
        CommonMethods.displaySnackBar('Unable to get location', context);
        return;
      }

      final initialLatLng = LatLng(loc.latitude!, loc.longitude!);
      String? address = await _reverseGeocode(initialLatLng);

      setState(() {
        _pickupLocation = initialLatLng;
        pickupController.text = address ??
            '${_pickupLocation!.latitude}, ${_pickupLocation!.longitude}';
      });

      if (_mapController != null && _pickupLocation != null) {
        _animateToLocation(_pickupLocation!);
      }

      _locationSubscription =
          _location.onLocationChanged.listen((locationData) {
        if (locationData.latitude != null &&
            locationData.longitude != null &&
            _mapController != null &&
            !_awaitingDropoff) {
          final newLocation =
              LatLng(locationData.latitude!, locationData.longitude!);
          _mapController!.animateCamera(CameraUpdate.newLatLng(newLocation));
        }
      });
    } catch (e) {
      CommonMethods.displaySnackBar('Location error: $e', context);
    }
  }

  void _onMapCreated(GoogleMapController controller) {
    _mapController = controller;
    _mapController!.setMapStyle(_getMapStyle());

    if (_pickupLocation != null) {
      _animateToLocation(_pickupLocation!);
    }

  void _animateToLocation(LatLng location, {double zoom = 16}) {
    if (_mapController != null) {
      _mapController!.animateCamera(
        CameraUpdate.newCameraPosition(
          CameraPosition(
            target: location,
            zoom: zoom,
            bearing: 30,
            tilt: 45,
          ),
        ),
      );
    }
  }

  void _onTap(LatLng position) async {
    if (!_awaitingDropoff) {
      setState(() {
        _pickupLocation = position;
        _awaitingDropoff = true;
      });
      final addr = await _reverseGeocode(position);
      if (addr != null) {
        setState(() {
          pickupController.text = addr;
        });
      }
    } else {
      setState(() {
        _dropoffLocation = position;
        _awaitingDropoff = false;
      });
      final addr = await _reverseGeocode(position);
      if (addr != null) {
        setState(() {
          dropoffController.text = addr;
        });
      }
    }
  }

  Set<Marker> _buildMarkers() {
    final markers = <Marker>{};
    if (_pickupLocation != null) {
      markers.add(Marker(
        markerId: const MarkerId('pickup'),
        position: _pickupLocation!,
        infoWindow: const InfoWindow(title: 'Pickup Location'),
        icon: BitmapDescriptor.defaultMarkerWithHue(BitmapDescriptor.hueGreen),

    }
    if (_dropoffLocation != null) {
      markers.add(Marker(
        markerId: const MarkerId('dropoff'),
        position: _dropoffLocation!,
        infoWindow: const InfoWindow(title: 'Destination'),
        icon: BitmapDescriptor.defaultMarkerWithHue(BitmapDescriptor.hueRed),

      markers.add(_driverMarker!);
    }
    return markers;
  }

  Future<void> _confirmRideWithVehicle(Map<String, dynamic> vehicleType) async {
    if (_pickupLocation == null || _dropoffLocation == null) return;
    setState(() => _requesting = true);

    final pickup = {
      'lat': _pickupLocation!.latitude,
      'lng': _pickupLocation!.longitude
    };
    final dropoff = {
      'lat': _dropoffLocation!.latitude,
      'lng': _dropoffLocation!.longitude
    };

    double baseFare = simpleFare(
      LatLng(pickup['lat']!, pickup['lng']!),
      LatLng(dropoff['lat']!, dropoff['lng']!),
    );

    double fare = baseFare * vehicleType["fareMultiplier"];

    try {
      final id = await _fs.createRideRequest(
        riderId: 'rider_test_1',
        pickup: pickup,
        dropoff: dropoff,
        fare: fare,
        vehicleType: vehicleType["name"],
      );
      setState(() {
        currentRideId = id;
        _fare = fare;
      });
      _showRideStatusModal();
    } catch (e) {
      ScaffoldMessenger.of(context)
          .showSnackBar(SnackBar(content: Text('Error creating ride: $e')));
    } finally {
      setState(() => _requesting = false);
    }
  }

  Future<void> _completeRide() async {
    if (currentRideId == null) return;
    await _fs.completeRide(rideId: currentRideId!);
    setState(() {
      currentRideId = null;
      _isPaymentCompleted = false;
    });
  }

  Future<LatLng?> _geocodeAddress(String address) async {
    try {
      final results = await geocoding.locationFromAddress(address);
      if (results.isNotEmpty) {
        return LatLng(results.first.latitude, results.first.longitude);
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Error finding location: $e')),
      );
    }
    return null;
  }

  Future<String?> _reverseGeocode(LatLng position) async {
    try {
      final results = await geocoding.placemarkFromCoordinates(
        position.latitude,
        position.longitude,
      );
      if (results.isNotEmpty) {
        final place = results.first;
        return "${place.street}, ${place.locality}, ${place.country}";
      }
    } catch (e) {
      print("Reverse geocoding error: $e");
    }
    return null;
  }

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:uuid/uuid.dart';
import 'package:eats/models/ride_model.dart';

class FirebaseService {
  final FirebaseFirestore _db = FirebaseFirestore.instance;
  final String rideCollection = 'rides';
  final _uuid = Uuid();

  // Create ride request
// Enhanced FirebaseService methods with error handling
  Future<String> createRideRequest({
    required String riderId,
    required Map<String, dynamic> pickup,
    required Map<String, dynamic> dropoff,
    required double fare,
    required String vehicleType,
  }) async {
    try {
      final id = _uuid.v4();
      final now = FieldValue.serverTimestamp();

      final Map<String, dynamic> data = {
        'id': id,
        'riderId': riderId,
        'driverId': null,
        'pickup': pickup,
        'dropoff': dropoff,
        'status': 'requested',
        'fare': fare,
        'vehicleType': vehicleType,
        'paymentStatus': 'unpaid',
        'paymentRequestId': null,
        'paymentProvider': null,
        'paymentRequestedAt': null,
        'paymentConfirmedAt': null,
        'createdAt': now,
        'updatedAt': now,
      };

      await _db.collection(rideCollection).doc(id).set(data);
      return id;
    } catch (e) {
      print('Error creating ride request: $e');
      throw Exception('Failed to create ride request. Please try again.');
    }
  }

  // Listen to available requests (for drivers)
  Stream<List<Ride>> streamAvailableRequests() {
    return _db
        .collection(rideCollection)
        .where('status', isEqualTo: 'requested')
        .snapshots()
        .map((snap) => snap.docs
            .map((d) => Ride.fromMap(Map<String, dynamic>.from(d.data())))
            .toList());
  }

  // Listen to a single ride
  Stream<Ride> streamRide(String id) {
    return _db
        .collection(rideCollection)
        .doc(id)
        .snapshots()
        .where((snap) => snap.exists)
        .map((snap) => Ride.fromMap(Map<String, dynamic>.from(snap.data()!)));
  }

  // Driver accepts ride
  Future<void> acceptRide(
      {required String rideId, required String driverId}) async {
    await _db.collection(rideCollection).doc(rideId).update({
      'driverId': driverId,
      'status': 'accepted',
      'updatedAt': FieldValue.serverTimestamp(),
    });
  }

  // Start ride
  Future<void> startRide({required String rideId}) async {
    await _db.collection(rideCollection).doc(rideId).update({
      'status': 'in_progress',
      'updatedAt': FieldValue.serverTimestamp(),
    });
  }

  // Complete ride
  Future<void> completeRide({required String rideId}) async {
    final doc = await _db.collection(rideCollection).doc(rideId).get();
    if (!doc.exists) return;
    final data = doc.data()!;
    // copy to history
    await _db.collection('rides_history').doc(rideId).set(data);
    // delete from active
    await _db.collection(rideCollection).doc(rideId).delete();
  }

  Future<void> updateRide(String rideId, Map<String, dynamic> data) async {
    try {
      final updates = Map<String, dynamic>.from(data);
      updates['updatedAt'] = FieldValue.serverTimestamp();
      await _db.collection(rideCollection).doc(rideId).update(updates);
    } catch (e) {
      print('Error updating ride $rideId: $e');
      rethrow;
    }
  }

  // Update driver location (simple implementation)
  Future<void> updateDriverLocation({
    required String driverId,
    required double lat,
    required double lng,
  }) async {
    await _db.collection('drivers').doc(driverId).set({
      'location': {'lat': lat, 'lng': lng},
      'timestamp': DateTime.now().toIso8601String()
    });
  }

  // Stream driver location
  Stream<Map<String, dynamic>?> streamDriverLocation(String driverId) {
    return _db.collection('drivers').doc(driverId).snapshots().map((s) =>
        s.exists ? Map<String, dynamic>.from(s.data()!['location']) : null);
  }

  // --- Payment helpers (used by your callable function / webhook on the backend)

  /// Called by your server/cloud-function immediately after sending the provider request.
  /// Marks the ride's payment as 'requested' and stores the provider request id.
  Future<void> markPaymentRequested({
    required String rideId,
    required String paymentRequestId,
    required String provider, // e.g. 'mtn_momo'
  }) async {
    await _db.collection(rideCollection).doc(rideId).update({
      'paymentStatus': 'requested',
      'paymentRequestId': paymentRequestId,
      'paymentProvider': provider,
      'paymentRequestedAt': FieldValue.serverTimestamp(),
      'updatedAt': FieldValue.serverTimestamp(),
    });
  }

  /// Called by the webhook handler when provider confirms or fails.
  Future<void> markPaymentResult({
    required String rideId,
    required bool success,
    String? providerRef,
  }) async {
    final updates = <String, dynamic>{
      'paymentStatus': success ? 'paid' : 'failed',
      'paymentConfirmedAt': FieldValue.serverTimestamp(),
      'updatedAt': FieldValue.serverTimestamp(),
    };
    if (providerRef != null) updates['paymentProviderRef'] = providerRef;
    await _db.collection(rideCollection).doc(rideId).update(updates);
  }
}

import 'package:location/location.dart';

class LocationService {
  final Location _location = Location();
  GoogleMapController? _mapController;

  LatLng? _pickupLocation;
  LatLng? _dropoffLocation;
  LatLng? _driverLocation;
  Marker? _driverMarker;

  LatLng? get currentLocation => _pickupLocation;
  LatLng? get pickupLocation => _pickupLocation;
  LatLng? get dropoffLocation => _dropoffLocation;

  Future<void> initializeLocation() async {
    bool serviceEnabled = await _location.serviceEnabled();
    if (!serviceEnabled) {
      serviceEnabled = await _location.requestService();
      if (!serviceEnabled) return;
    }

    PermissionStatus permissionGranted = await _location.hasPermission();
    if (permissionGranted == PermissionStatus.denied) {
      permissionGranted = await _location.requestPermission();
      if (permissionGranted != PermissionStatus.granted) return;
    }

    final loc = await _location.getLocation();
    _pickupLocation =
        LatLng(loc.latitude ?? -26.2041, loc.longitude ?? 28.0473);

    _location.onLocationChanged.listen((locationData) {
      if (locationData.latitude != null &&
          locationData.longitude != null &&
          _mapController != null) {
        final newLocation =
            LatLng(locationData.latitude!, locationData.longitude!);
        _mapController!.animateCamera(CameraUpdate.newLatLng(newLocation));
      }
    });
  }

  void onMapCreated(GoogleMapController controller) {
    _mapController = controller;
    if (_pickupLocation != null) {
      _mapController!.animateCamera(
        CameraUpdate.newCameraPosition(
          CameraPosition(target: _pickupLocation!, zoom: 14),
        ),
      );
    }
  }

  void updatePickupLocation(LatLng location) {
    _pickupLocation = location;
  }

  void updateDropoffLocation(LatLng location) {
    _dropoffLocation = location;
  }

  void updateDriverLocation(LatLng newLocation) {
    _driverLocation = newLocation;
    _driverMarker = Marker(
      markerId: MarkerId('driver'),
      position: newLocation,
      infoWindow: InfoWindow(title: 'Driver'),
      icon: BitmapDescriptor.defaultMarkerWithHue(BitmapDescriptor.hueBlue),
    );

    _mapController?.animateCamera(CameraUpdate.newLatLng(newLocation));
  }

  Set<Marker> buildMarkers() {
    final markers = <Marker>{};
    if (_pickupLocation != null) {
      markers.add(Marker(
        markerId: MarkerId('pickup'),
        position: _pickupLocation!,
        infoWindow: InfoWindow(title: 'Pickup'),
        icon: BitmapDescriptor.defaultMarkerWithHue(BitmapDescriptor.hueGreen),
      ));
    }
    if (_dropoffLocation != null) {
      markers.add(Marker(
        markerId: MarkerId('dropoff'),
        position: _dropoffLocation!,
        infoWindow: InfoWindow(title: 'Dropoff'),
        icon: BitmapDescriptor.defaultMarkerWithHue(BitmapDescriptor.hueRed),
      ));
    }
    if (_driverMarker != null) markers.add(_driverMarker!);
    return markers;
  }

  Future<String?> reverseGeocode(LatLng position) async {
    try {
      final results = await geocoding.placemarkFromCoordinates(
          position.latitude, position.longitude);
      if (results.isNotEmpty) {
        final place = results.first;
        return "${place.street}, ${place.locality}, ${place.country}";
      }
    } catch (e) {
      print("Reverse geocoding error: $e");
    }
    return null;
  }

  Future<LatLng?> geocodeAddress(String address) async {
    try {
      final results = await geocoding.locationFromAddress(address);
      if (results.isNotEmpty) {
        return LatLng(results.first.latitude, results.first.longitude);
      }
    } catch (e) {
      print("Geocoding error: $e");
    }
    return null;
  }
}

import 'package:flutter/material.dart';

/// A rainbow rotating border that wraps an ElevatedButton.
/// Supports both sync and async onPressed.
class RainbowButton extends StatefulWidget {
  final Future<void> Function()? onPressed; // now supports async!
  final Widget child;
  final double borderWidth;
  final double borderRadius;
  final Duration duration;
  final Color backgroundColor;

  const RainbowButton({
    super.key,
    required this.onPressed,
    required this.child,
    this.borderWidth = 3,
    this.borderRadius = 12,
    this.duration = const Duration(seconds: 6),
    this.backgroundColor = Colors.black,
  });

  @override
  State<RainbowButton> createState() => _RainbowButtonState();
}

class _RainbowButtonState extends State<RainbowButton>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  bool _loading = false;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(vsync: this, duration: widget.duration)
      ..repeat();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  Future<void> _handlePress() async {
    if (widget.onPressed == null) return;
    setState(() => _loading = true);
    try {
      await widget.onPressed!();
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (_, __) {
        return Container(
          padding: EdgeInsets.all(widget.borderWidth),
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(widget.borderRadius),
            gradient: SweepGradient(
              colors: const [
                Colors.red,
                Colors.orange,
                Colors.yellow,
                Colors.green,
                Colors.cyan,
                Colors.blue,
                Colors.purple,
                Colors.red,
              ],
              stops: const [0.0, 0.16, 0.33, 0.5, 0.66, 0.83, 1.0, 1.0],
              transform: GradientRotation(_controller.value * 6.283),
            ),
          ),
          child: ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: widget.backgroundColor,
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(
                    widget.borderRadius - widget.borderWidth),
              ),
              padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 14),
              elevation: 0,
            ),
            onPressed:
                widget.onPressed == null || _loading ? null : _handlePress,
            child: _loading
                ? const SizedBox(
                    height: 10,
                    width: 10,
                    child: CircularProgressIndicator(
                      strokeWidth: 2,
                      color: Colors.white,
                    ),
                  )
                : widget.child,
          ),
        );
      },
    );
  }
}

const functions = require("firebase-functions");
const admin = require("firebase-admin");

const { v4: uuidv4 } = require("uuid");

admin.initializeApp();

const MOMO_SUBSCRIPTION_KEY = functions.config().momo.primarykey; // from firebase config
const MOMO_USER_ID = functions.config().momo.userid;
const MOMO_API_KEY = functions.config().momo.apikey;
const MOMO_ENV = functions.config().momo.env || "sandbox";
const MOMO_API = MOMO_ENV === "sandbox"
  ? "https://sandbox.momodeveloper.mtn.com"
  : "https://momodeveloper.mtn.com";

// Step 1: Get access token
async function getMomoAccessToken() {
  const auth = Buffer.from(`${MOMO_USER_ID}:${MOMO_API_KEY}`).toString("base64");
  const response = await axios.post(`${MOMO_API}/collection/token/`, {}, {
    headers: {
      "Authorization": `Basic ${auth}`,
      "Ocp-Apim-Subscription-Key": MOMO_SUBSCRIPTION_KEY
    }
  });
  return response.data.access_token;
}

// Step 2: Request payment from customer
exports.requestMomoPayment = functions.https.onCall(async (data, context) => {
  const { phone, amount, rideId } = data;

  if (!phone || !amount || !rideId) {
    throw new functions.https.HttpsError("invalid-argument", "phone, amount, and rideId are required");
  }

  const token = await getMomoAccessToken();
  const referenceId = uuidv4();

  await axios.post(`${MOMO_API}/collection/v1_0/requesttopay`, {
    amount: amount.toString(),
    currency: "SZL",
    externalId: rideId,
    payer: { partyIdType: "MSISDN", partyId: phone },
    payerMessage: "Payment for Kombi ride",
    payeeNote: "Thank you for riding with us"
  }, {
    headers: {
      "Authorization": `Bearer ${token}`,
      "X-Reference-Id": referenceId,
      "X-Target-Environment": MOMO_ENV,
      "Ocp-Apim-Subscription-Key": MOMO_SUBSCRIPTION_KEY,
      "Content-Type": "application/json"
    }
  });

  // Optionally, store referenceId to track payment status
  await admin.firestore().collection("rides").doc(rideId).set({
    momoReferenceId: referenceId,
    momoAmount: amount,
    momoPhone: phone,
    momoStatus: "PENDING"
  }, { merge: true });

  return { referenceId };
});

// Verify MoMo Payment
exports.verifyMomoPayment = functions.https.onCall(async (data, context) => {
  const { referenceId, rideId } = data;

  if (!referenceId || !rideId) {
    throw new functions.https.HttpsError("invalid-argument", "referenceId and rideId are required");
  }

  const token = await getMomoAccessToken();

  try {
    const response = await axios.get(`${MOMO_API}/collection/v1_0/requesttopay/${referenceId}`, {
      headers: {
        "Authorization": `Bearer ${token}`,
        "X-Target-Environment": MOMO_ENV,
        "Ocp-Apim-Subscription-Key": MOMO_SUBSCRIPTION_KEY
      }
    });

    const status = response.data.status;

    // Update Firestore based on payment status
    let paid = false;
    if (status === "SUCCESSFUL") paid = true;

    await admin.firestore().collection("rides").doc(rideId).set({
      momoStatus: status,
      paid: paid,
      paymentMethod: "MoMo",
      momoDetails: response.data
    }, { merge: true });

    return { success: paid, status };
  } catch (error) {
    console.error("MoMo verification error:", error.response?.data || error.message);
    throw new functions.https.HttpsError("internal", "Failed to verify MoMo payment");
  }
});
